4. OBTENER CITAS DEL PACIENTE (Mejorar el existente)
--------------------------------------------------------------------------------

Endpoint:
GET /api/citas/paciente/{idUsuario}

Query SQL:
SELECT 
    c.id_cita,
    c.fecha_cita AS fecha,
    c.hora_cita AS hora,
    c.estado,
    c.motivo_consulta,
    CONCAT(p.nombre1, ' ', p.apellido_paterno) AS medico_nombre,
    e.nombre AS especialidad,
    se.nombre AS sub_especialidad,
    dc.precio_total AS precio,
    hb.hora_inicio,
    hb.hora_fin
FROM cita c
JOIN paciente pac ON c.id_paciente = pac.id_paciente
JOIN usuario u ON pac.id_persona = (SELECT id_persona FROM persona WHERE id_persona = u.id_persona)
JOIN detalle_cita dc ON c.id_detalle_cita = dc.id_detalle_cita
JOIN medico_especialidad me ON dc.id_medico_especialidad = me.id_medico_especialidad
JOIN medico m ON me.id_medico = m.id_medico
JOIN persona p ON m.id_persona = p.id_persona
JOIN especialidad e ON me.id_especialidad = e.id_especialidad
LEFT JOIN sub_especialidad se ON dc.id_subespecialidad = se.id_subespecialidad
LEFT JOIN horario_bloque hb ON c.id_disponibilidad = hb.id_disponibilidad
WHERE u.id_usuario = ?
ORDER BY c.fecha_cita DESC, c.hora_cita DESC;

6. OBTENER PAGOS DEL PACIENTE (NUEVO - FALTA IMPLEMENTAR)
--------------------------------------------------------------------------------

Endpoint:
GET /api/pagos/paciente/{idUsuario}

Query SQL:
SELECT 
    p.id_pago,
    p.monto_total,
    p.metodo AS metodo_pago,
    p.estado,
    p.fecha_pago,
    c.id_cita,
    c.fecha_cita,
    c.hora_cita,
    CONCAT(per.nombre1, ' ', per.apellido_paterno) AS medico_nombre,
    e.nombre AS especialidad,
    se.nombre AS sub_especialidad,
    c.motivo_consulta
FROM pago p
JOIN pago_detalle pd ON p.id_pago = pd.id_pago
JOIN cita c ON pd.id_cita = c.id_cita
JOIN paciente pac ON c.id_paciente = pac.id_paciente
JOIN usuario u ON pac.id_persona = (SELECT id_persona FROM persona WHERE id_persona = u.id_persona)
JOIN detalle_cita dc ON c.id_detalle_cita = dc.id_detalle_cita
JOIN medico_especialidad me ON dc.id_medico_especialidad = me.id_medico_especialidad
JOIN medico m ON me.id_medico = m.id_medico
JOIN persona per ON m.id_persona = per.id_persona
JOIN especialidad e ON me.id_especialidad = e.id_especialidad
LEFT JOIN sub_especialidad se ON dc.id_subespecialidad = se.id_subespecialidad
WHERE u.id_usuario = ?
ORDER BY p.fecha_pago DESC;
